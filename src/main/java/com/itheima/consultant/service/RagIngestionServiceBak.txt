import com.itheima.consultant.dto.CodeChunk;
import dev.langchain4j.data.document.Document;
import dev.langchain4j.data.segment.TextSegment;

@Service
public class RagIngestionServiceBak {

    private static final Logger logger = LoggerFactory.getLogger(RagIngestionService.class);

    @Autowired
    private EmbeddingModel embeddingModel; // 用于将文本转换为向量

    @Autowired
    private EmbeddingStore<TextSegment> embeddingStore; // 用于存储向量和文本

    private static final List<String> SUPPORTED_EXTENSIONS = List.of(".java", ".js", ".py", ".md", ".txt", ".html", ".css", ".xml", ".yml", ".properties");
    @Autowired
    private HttpSession httpSession;

    /**
     * 公开的核心方法，用于接收文件和会话ID，并执行整个注入流程
     * @param file 用户上传的 .zip 文件
     * @param sessionId 唯一的会话ID
     */
    public void ingest(MultipartFile file, String sessionId) throws IOException {
        logger.info("开始为会话 '{}' 处理文件 '{}'", sessionId, file.getOriginalFilename());

        // 步骤1：解压文件、解析内容、分块并打上标签
        List<CodeChunk> chunks = unzipAndParse(file.getInputStream(), sessionId);

        // 步骤2：将分块后的代码进行向量化并存入Redis
        embedAndStore(chunks);

        logger.info("会话 '{}' 的文件处理和知识注入完成", sessionId);
    }

    /**
     * 负责解压、解析和分块的核心逻辑
     * @param inputStream zip文件的输入流
     * @param sessionId 会话ID，用于给每个块打上标签
     * @return 包含了所有代码块的列表
     * @throws IOException 如果读取流时发生错误
     */
    private List<CodeChunk> unzipAndParse(InputStream inputStream, String sessionId) throws IOException {
        //创建一个接受器，接受zip文件里面的内容
        List<CodeChunk> chunks = new ArrayList<>();

        // 使用 try-with-resources 语法，它能保证无论是否发生异常，
        // ZipInputStream 都会被自动关闭，比手动 finally 关闭更简洁安全。
        //这里面的ZipInputStream zis = new ZipInputStream(inputStream)的含义是接受输入流，转换成zip里面的一个个文件的流
        try (ZipInputStream zis = new ZipInputStream(inputStream)) {

            ZipEntry zipEntry;//为什么不是ZipEntry zipEntry=new ZipEntry();
            // 这是遍历zip包内所有文件的标准写法
            while ((zipEntry = zis.getNextEntry()) != null) {

                // 1. 跳过目录，只处理文件,isDirectory()又是哪里来的，zip文件里面全是pdf的话也有这个目录吗
                if (zipEntry.isDirectory()) {
                    continue;
                }

                String fileName = zipEntry.getName();

                // 2. 【安全检查】防止"Zip Slip"路径遍历攻击
                // 这是一个非常重要的安全措施！
                Path entryPath = Paths.get(fileName).normalize();
                // 检查规范化后的路径是否以".."开头，如果是，则意味着它试图访问父目录，有安全风险
                if (entryPath.startsWith("..")) {
                    logger.warn("检测到不安全的ZIP条目，已跳过: {}", fileName);
                    continue;
                }

                // 3. 判断是否是支持处理的文件类型
                if (isSupportedFile(fileName)) {

                    // 4. 读取文件内容（我们接下来会详细解释这部分）
                    // 这里的zis已经是当前zipEntry的输入流了，不需要再额外操作
                    //为什么还要经过new InputStreamReader再转换一次，直接把zis放进去不星吗
                    //不可以，不进行这一步操作的话，可能zis是gbk的格式，你在utf8的文件读取不了
                    String content = new BufferedReader(
                            new InputStreamReader(zis, StandardCharsets.UTF_8))
                            .lines()
                            .collect(Collectors.joining("\n"));

                    if (content.trim().isEmpty()) {
                        continue;
                    }

                    // 5. 调用分块方法，并加入列表
                    List<CodeChunk> fileChunks = chunkContent(content, fileName, sessionId);
                    chunks.addAll(fileChunks);
                }
            }
        }
        return chunks;
    }

    /**
     * 负责将代码块向量化并存入向量数据库
     * @param chunks 代码块列表
     */
    private void embedAndStore(List<CodeChunk> chunks) {

        if (chunks == null || chunks.isEmpty()) {
            logger.info("没有要注入的代码块，流程结束。");
            return;
        }

        // 步骤1：将我们自己的 List<CodeChunk> 转换成 LangChain4j 的 List<TextSegment>
        List<TextSegment> segments = new ArrayList<>();
        for (CodeChunk chunk : chunks) {
            // TextSegment.from() 方法需要文本和元数据两个参数
            segments.add(TextSegment.from(chunk.getContent(), chunk.getMetadata()));
        }


        embeddingStore.addAll(segments);

        logger.info("成功注入了 {} 个代码块到向量数据库。", segments.size());
        // TODO: 在这里实现向量化和存储的逻辑
        // 提示：
        // 1. 将 List<CodeChunk> 转换成 LangChain4j 的 List<TextSegment>。
        //    TextSegment 同时包含文本和元数据，正好对应我们的 CodeChunk。
        // 2. 使用注入的 embeddingModel 对 List<TextSegment> 进行向量化。
        // 3. 使用注入的 embeddingStore 将向量化后的结果存入。
    }

    // --- 辅助方法 ---
    private List<CodeChunk> chunkContent(String content, String filePath, String sessionId) {
        // 创建一个分块器，目标每个块500个字符，块之间重叠100个字符
        // 重叠部分有助于在检索时，不会因为切分而丢失上下文
        DocumentSplitter splitter = DocumentSplitters.recursive(500, 100);

        // 使用分块器切分文本,Document.from的作用把String 的content包装为Document，这有助于它携带文档级别的元数据
        List<TextSegment> textChunks = splitter.split(Document.from(content));

        List<CodeChunk> codeChunks = new ArrayList<>();
        for (TextSegment textChunk : textChunks) {
            Map<String, Object> metadata = new HashMap<>();
            metadata.put("source", filePath);
            metadata.put("sessionId", sessionId); // 为每个小块都打上标签！
            codeChunks.add(new CodeChunk(textChunk.text(), metadata));
        }

        logger.info("为文件 '{}' 创建了 {} 个代码块", filePath, codeChunks.size());
        return codeChunks;
    }

    private boolean isSupportedFile(String fileName) {
        return SUPPORTED_EXTENSIONS.stream().anyMatch(fileName::endsWith);
    }
}
